# 炎魔法発射後の燃える床の手順

設計: [炎魔法発射後の燃える床の設計](炎魔法発射後の燃える床の設計.md)
経緯: [魔法と弾の経緯](魔法と弾の経緯.md)

炎の弾が着弾したところの床を、しばらく燃やし続けるまでの手順です。

**出来上がるもの:** 炎の弾が着弾すると、真下の床が中心から一気に燃え広がり、4秒間燃え続けます。燃えている範囲に入った敵対する相手は、燃焼状態になります。溜め具合が大きいほど、燃える範囲が広くなります。

**前提:**

- Visual Studio で `Action/Game/Game.sln` を開き、構成は **Debug / x64** でビルドする。
- 先に [炎魔法の手順](炎魔法の手順.md) を済ませておく（順番は [README の「ゲームを作る順番」](../../README.md#ゲームを作る順番)）。燃やすのは [状態異常の手順](../キャラクター/状態異常の手順.md) で作った `StatusEffects::Burn()`。
- エフェクト `Assets/effect/fire_ground.efk` は、リポジトリにもとからある。

| 手順 | 足すもの | 確かめ方 |
|---|---|---|
| 1 | 炎魔法発射後の燃える床 `BurningGround` | ビルドが通る |
| 2 | 炎の弾が着弾したら、真下の床を燃やす | 着弾したところの床が燃える |

**読み方:**

- 足す場所は「〇行目 `その行に書いてあること` の下」で示します。行番号は、それより前の小手順を済ませたときの番号です。
- 枠の左の数字は、足したあとの行番号です。右の枠のコードを、そのまま書きます。
- 書き換えるところは、「前」と「後」の枠を並べています。
- VSCode で `Ctrl+Shift+V` を押してプレビューで読みます。

	// fire_ground.efk は半径10の円に炎を出す。Scale 10 倍でワールドの半径100になる。
	constexpr float EFFECT_RADIUS_AT_SCALE_ONE = 10.0f;

	constexpr float BURN_SECONDS = 1.5f;     // 範囲に入った相手を燃やす時間。範囲から出ても、この時間は燃え続ける。
	constexpr float BURN_HEIGHT = 150.0f;    // 床からこの高さまでに足元がある相手を燃やす（ジャンプで飛び越えられる）。
	constexpr float BURN_DEPTH = 50.0f;      // 床よりこれだけ下に足元がある相手も燃やす（坂の下側など）。
	constexpr float TARGET_RADIUS_RATE = 0.5f;  // 相手の当たり判定の半径のうち、範囲の判定に足す割合。
	constexpr float SPREAD_SECONDS = 0.4f;     // 中心から最大の広さまで燃え広がる秒数。着弾の飛び散りに合わせる。
	constexpr float SPREAD_START_RATE = 0.1f;   // 燃え始めの広さ（最大に対する割合）。
}

BurningGround::~BurningGround()
{
	// Effect のデストラクタは再生を止めないため、ここで明示的に止める。
	if (!m_isValid || EffectEngine::GetInstance() == nullptr) {
		return;
	}
	m_effect.Stop();
}

void BurningGround::Init(const Vector3& position, float radius, float seconds, const IGameObject* owner, Faction faction)
{
	m_position = position;
	m_radius = radius;
	m_seconds = seconds;
	m_owner = owner;
	m_faction = faction;
}

bool BurningGround::Start()
{
	m_isValid = FireBullet::RegisterEffects();
	if (!m_isValid) {
		return true;
	}

	m_effect.Init(GROUND_EFFECT_ID);
	m_effect.Play();
	m_effect.SetPosition(m_position);
	UpdateEffectScale();
	return true;
}

void BurningGround::Update()
{
	if (!m_isValid) {
		DeleteGO(this);
		return;
	}

	switch (m_state) {
	case EnState::Burning:
		if (!m_isSpreadFinished) {
			// 燃え広がっている間は、出ている炎ごと外へ広げる。最大の広さに達したフレームまで更新する。
			UpdateEffectScale();
			m_isSpreadFinished = m_elapsed >= SPREAD_SECONDS;
		}
		BurnTargetsInRange();
		m_elapsed += g_gameTime->GetFrameDeltaTime();
		if (m_elapsed >= m_seconds) {
			// 炎の発生だけを止め、出ている炎は寿命まで燃やして自然に消す。
			m_effect.StopRoot();
			m_state = EnState::Vanishing;
		}
		break;
	case EnState::Vanishing:
		if (!m_effect.IsPlay()) {
			DeleteGO(this);
		}
		break;
	}
}

void BurningGround::BurnTargetsInRange()
{
	for (HitTarget* target : HitTargetRegistry::GetTargets()) {
		if (!target->IsActive() || target->GetOwner() == m_owner) {
			continue;
		}
		if (!IsHostile(m_faction, target->GetFaction())) {
			continue;
		}
		StatusEffects* statusEffects = target->GetStatusEffects();
		if (statusEffects == nullptr) {
			continue;
		}

		const Vector3& foot = target->GetFootPosition();
		const float height = foot.y - m_position.y;
		if (height < -BURN_DEPTH || height > BURN_HEIGHT) {
			continue;
		}
		const float dx = foot.x - m_position.x;
		const float dz = foot.z - m_position.z;
		const float range = GetCurrentRadius() + target->GetRadius() * TARGET_RADIUS_RATE;
		if (dx * dx + dz * dz > range * range) {
			continue;
		}
		statusEffects->Burn(BURN_SECONDS);
	}
}

float BurningGround::GetCurrentRadius() const
{
	if (m_elapsed >= SPREAD_SECONDS) {
		return m_radius;
	}
	// 最初に勢いよく広がり、最大に近づくにつれてゆっくりになる。
	const float t = m_elapsed / SPREAD_SECONDS;
	const float easeOut = 1.0f - (1.0f - t) * (1.0f - t);
	return m_radius * (SPREAD_START_RATE + (1.0f - SPREAD_START_RATE) * easeOut);
}

void BurningGround::UpdateEffectScale()
{
	const float scale = GetCurrentRadius() / EFFECT_RADIUS_AT_SCALE_ONE;
	m_effect.SetScale(Vector3(scale, scale, scale));
	m_effect.Update();
}

#include "stdafx.h"
#include "FireBullet.h"
#include "BurningGround.h"
#include "EffectId.h"
#include "HitTarget.h"

namespace
{
	// エフェクトは Effekseer 用に自作したもの(Assets/effect/fire_*.efkproj が編集元)。
	// テクスチャは Effekseer サンプル(01_Pierre01, CC-0)のものを使用。
	constexpr int BULLET_EFFECT_ID = static_cast<int>(EffectId::FireBullet);
	constexpr int HIT_EFFECT_ID = static_cast<int>(EffectId::FireHit);

	struct EffectEntry
	{
		EffectId id;
		const char16_t* path;
	};
	// 炎魔法で使うエフェクトの一覧。FireMagic・BurningGround もこの登録を使う。
	constexpr EffectEntry EFFECTS[] = {
		{ EffectId::FireBullet, u"Assets/effect/fire_bullet.efk" },
		{ EffectId::FireCharge, u"Assets/effect/fire_charge.efk" },
		{ EffectId::FireCharged, u"Assets/effect/fire_charged.efk" },
		{ EffectId::FireHit, u"Assets/effect/fire_hit.efk" },
		{ EffectId::FireGround, u"Assets/effect/fire_ground.efk" },
	};

	// 溜めなし(0)〜最大まで溜めた(1)ときの値。間は線形に補間する。
	constexpr float MIN_RADIUS = 14.0f;               // 弾の当たり判定の半径。
	constexpr float MAX_RADIUS = 45.0f;
	constexpr float MIN_BULLET_EFFECT_SCALE = 8.0f;     // 弾のエフェクトの拡大率。
	constexpr float MAX_BULLET_EFFECT_SCALE = 26.0f;
	constexpr float MIN_HIT_EFFECT_SCALE = 12.0f;       // 着弾エフェクトの拡大率。
	constexpr float MAX_HIT_EFFECT_SCALE = 26.0f;
	constexpr float MIN_GROUND_RADIUS = 130.0f;        // 炎魔法発射後の燃える床の半径。
	constexpr float MAX_GROUND_RADIUS = 300.0f;

	constexpr float SPEED = 1300.0f;
	constexpr float RANGE = 2000.0f;
	constexpr float GROUND_BURN_SECONDS = 4.0f;        // 床が燃えている時間。

	// 炎魔法発射後の燃える床を置く場所を探すレイ。着弾点の少し手前・少し上から真下へ飛ばす。
	constexpr float FLOOR_RAY_BACK_OFF = 20.0f;         // 壁に当たったとき、壁にめり込まないよう手前に戻す距離。
	constexpr float FLOOR_RAY_LIFT = 10.0f;
	constexpr float FLOOR_SEARCH_DEPTH = 400.0f;       // これより下に床がなければ床は燃えない。

	float Lerp(float rate, float a, float b)
	{
		return a + (b - a) * rate;
	}

	BulletSettings MakeSettings(float chargeRate)
	{
		BulletSettings settings;
		settings.speed = SPEED;
		settings.range = RANGE;
		settings.radius = Lerp(chargeRate, MIN_RADIUS, MAX_RADIUS);
		return settings;
	}

FireBullet::FireBullet()
	: BulletBase(MakeSettings(0.0f))
{
}

FireBullet::~FireBullet()
{
	// Effect のデストラクタは再生を止めないため、ここで明示的に止める。
	// 通常は両方とも再生が終わってから削除されるので何も起きない。
	if (!m_isEffectReady || EffectEngine::GetInstance() == nullptr) {
		return;
	}
	m_bulletEffect.Stop();
	m_hitEffect.Stop();
}

void FireBullet::SetCharge(float chargeRate)
{
	m_chargeRate = chargeRate < 0.0f ? 0.0f : (chargeRate > 1.0f ? 1.0f : chargeRate);
	SetSettings(MakeSettings(m_chargeRate));
}

bool FireBullet::RegisterEffects()
{
	EffectEngine* effectEngine = EffectEngine::GetInstance();
	if (effectEngine == nullptr) {
		K2_ASSERT(false, "EffectEngine が未生成のため炎魔法のエフェクトを再生できません。");
		return false;
	}

	// 登録済みの番号は内部で無視されるので、何度呼んでも読み込みは1回だけ。
	for (const EffectEntry& entry : EFFECTS) {
		const int id = static_cast<int>(entry.id);
		effectEngine->ResistEffect(id, entry.path);
		if (effectEngine->LoadEffect(id) == nullptr) {
			K2_ASSERT(false, "炎魔法のエフェクト(Assets/effect/fire_*.efk)の読み込みに失敗しました。");
			return false;
		}
	}
	return true;
}

Vector3 FireBullet::GetBulletEffectScale(float chargeRate)
{
	const float scale = Lerp(chargeRate, MIN_BULLET_EFFECT_SCALE, MAX_BULLET_EFFECT_SCALE);
	return Vector3(scale, scale, scale);
}

bool FireBullet::OnStart()
{
	if (!RegisterEffects()) {
		return false;
	}

	m_bulletEffect.Init(BULLET_EFFECT_ID);
	m_hitEffect.Init(HIT_EFFECT_ID);
	m_isEffectReady = true;

	// 新しいハンドルには行列が設定されていないので、再生直後に OnMove() で必ず設定される。
	m_bulletEffect.Play();
	return true;
}

void FireBullet::OnMove()
{
	m_bulletEffect.SetPosition(GetPosition());
	m_bulletEffect.SetRotation(GetRotation());
	m_bulletEffect.SetScale(GetBulletEffectScale(m_chargeRate));
	// 全体ではなくルートだけを動かし、炎と煙の尾を通った場所に残す。
	m_bulletEffect.UpdateRoot();
}

void FireBullet::OnHitTarget(HitTarget& target, const Vector3& hitPosition)
{
	(void)hitPosition;
	// 直撃では何もしない。相手は足元の炎魔法発射後の燃える床に入るので、そこで燃焼状態になる。
	m_hasHitTarget = true;
	m_hitFootPosition = target.GetFootPosition();
}

void FireBullet::OnExplode(const Vector3& hitPosition)
{
	// 弾本体は即座に消し、尾の炎と煙は寿命まで残す。
	m_bulletEffect.StopRoot();

	const float hitScale = Lerp(m_chargeRate, MIN_HIT_EFFECT_SCALE, MAX_HIT_EFFECT_SCALE);
	m_hitEffect.Play();
	m_hitEffect.SetPosition(hitPosition);
	m_hitEffect.SetScale(Vector3(hitScale, hitScale, hitScale));
	m_hitEffect.Update();

	SpawnBurningGround(hitPosition);
}

bool FireBullet::IsVanishFinished() const
{
	// 尾と着弾エフェクトが両方消え切ったら終わり。炎魔法発射後の燃える床は別のオブジェクトなので待たない。
	return !m_bulletEffect.IsPlay() && !m_hitEffect.IsPlay();
}

void FireBullet::SpawnBurningGround(const Vector3& hitPosition)
{
	PhysicsWorld* physicsWorld = PhysicsWorld::GetInstance();
	if (physicsWorld == nullptr) {
		return;
	}

	// 相手に当たったらその足元から、地形・射程切れなら着弾点の少し手前から、真下の床を探す。
	// 地形のレイはキャラクターのカプセルにも当たるので、相手に当たったときは足元から調べる。
	Vector3 rayStart = m_hasHitTarget ? m_hitFootPosition : hitPosition - GetDirection() * FLOOR_RAY_BACK_OFF;
	rayStart.y += FLOOR_RAY_LIFT;
	Vector3 rayEnd = rayStart;
	rayEnd.y -= FLOOR_SEARCH_DEPTH;

	Vector3 floorPosition;
	if (!physicsWorld->RayTest(rayStart, rayEnd, floorPosition)) {
		// 下に床がない（崖の外など）。床は燃やさない。
		return;
	}

	BurningGround* ground = NewGO<BurningGround>(1, BurningGround::OBJECT_NAME);
	ground->Init(
		floorPosition,
		Lerp(m_chargeRate, MIN_GROUND_RADIUS, MAX_GROUND_RADIUS),
		GROUND_BURN_SECONDS,
		GetOwner(),
		GetFaction());
}
