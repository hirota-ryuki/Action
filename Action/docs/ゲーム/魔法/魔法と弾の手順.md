# 魔法と弾の手順

作成日: 2026-11-08
設計: [魔法と弾の設計](魔法と弾の設計.md)

魔法の基底クラス `MagicBase` と、魔法を持って撃つ部品 `MagicCaster` を作り、キャラクターに持たせるまでの手順です。上から順にやると、[魔法と弾の設計](魔法と弾の設計.md) の形になります。弾（`BulletBase`）は [弾の手順](弾の手順.md)、それぞれの魔法は [氷魔法の手順](氷魔法の手順.md)・[炎魔法の手順](炎魔法の手順.md) で作ります。

**出来上がるもの:** Player と Enemy が `MagicCaster` を持ち、発射ボタン（`fire`・`subFire`）と切り替え（`magicSwitch`）の入力に合わせて、持っている魔法の関数を呼ぶようになります。まだ魔法を1つも持たせていないので、撃っても何も出ません。

**前提:**

- Visual Studio 2022 で `Action/Game/Game.sln` を開き、構成は **Debug / x64** でビルドする。
- ゲームの手順書は [README の「ゲームを作る順番」](../../README.md#ゲームを作る順番) の順に進める。この手順の前に、[走りから止まる手順](../プレイヤー/走りから止まる手順.md) まで済ませておく。
- 作業を始める前に、main からブランチを切る。

| 手順 | 足すもの | 確かめ方 |

| 1 | 魔法の基底クラス `MagicBase` | ビルドが通る |

<!-- new: ../../../Game/Source/Magic/MagicBase.h -->

```cpp
#pragma once

#include "Faction.h"

// 魔法を使うときに渡される情報。
struct MagicCastContext
{
	IGameObject* caster = nullptr;      // 使ったキャラクター。
	Faction faction = Faction::Neutral;
	Vector3 origin = Vector3::Zero;     // 撃ち出す位置。
	Vector3 direction = Vector3::AxisZ; // 撃ち出す向き（正規化済み）。
	float deltaTime = 0.0f;
};

// 魔法の基底クラス。「どう撃つか」を決める（押した瞬間に撃つ、溜めて離したら撃つ、押している間出し続ける、など）。
// 飛んでいく弾そのものは BulletBase の派生クラスが担当する。
// MagicCaster が発射ボタンの状態に応じて OnPress / OnHold / OnRelease を呼ぶ。
class MagicBase
{
public:
	virtual ~MagicBase() = default;

	// デバッグ表示やUI用の名前。
	virtual const char* GetName() const = 0;

	// 発射ボタンを押した瞬間。
	virtual void OnPress(const MagicCastContext& context) { (void)context; }
	// 発射ボタンを押している間、毎フレーム（押した瞬間のフレームも含む）。
	virtual void OnHold(const MagicCastContext& context) { (void)context; }
	// 発射ボタンを離した瞬間。
	virtual void OnRelease(const MagicCastContext& context) { (void)context; }
	// 押している途中で中断されたとき（凍結した、魔法を切り替えた、など）。溜めの破棄などに使う。
	virtual void OnCancel() {}

	// 毎フレーム呼ばれる。クールダウンを進める。
	virtual void Update(float deltaTime)
	{
		if (m_cooldown > 0.0f) {
			m_cooldown -= deltaTime;
		}
	}

	bool IsReady() const
	{
		return m_cooldown <= 0.0f;
	}

protected:
	void StartCooldown(float seconds)
	{
		m_cooldown = seconds;
	}

private:
	float m_cooldown = 0.0f;  // 再び使えるまでの残り秒数。
};
```

- **4つの関数は、何もしない中身を持たせておく。** 派生クラスは、使う関数だけを上書きします（氷魔法は `OnPress()` だけ、炎魔法は4つとも）。
- **クールダウンは基底クラスで進める。** 派生クラスは撃ったときに `StartCooldown()` を呼び、撃つ前に `IsReady()` を見るだけです。

**確かめ方:** ソリューションのビルドが通る。

<!-- new: ../../../Game/Source/Magic/MagicCaster.h -->

```cpp
#pragma once

#include "Faction.h"
#include "MagicBase.h"

struct CharacterInput;
struct ButtonState;

// キャラクターが持つ魔法の一覧と、今選んでいる魔法を管理する。
// 入力の発射ボタン・切り替えに応じて、魔法の OnPress / OnHold / OnRelease を呼ぶ。
class MagicCaster
{
public:
	struct Settings
	{
		float castHeight = 70.0f;   // 照準の指定がないとき、足元からこの高さで撃つ。
		float castForward = 60.0f;  // 自分のカプセルに当たらないよう、撃つ位置を前に出す距離。
	};

	void Init(IGameObject* owner, Faction faction, const Settings& settings);
	void SetFaction(Faction faction)
	{
		m_faction = faction;
	}
	void AddMagic(std::unique_ptr<MagicBase> magic);
	// サブの発射ボタン（subFire）で撃つ魔法。切り替えの対象にはならず、いつでも撃てる。
	void SetSubMagic(std::unique_ptr<MagicBase> magic);

	// canAct が false（凍結中など）の間は撃てない。押している途中なら中断する。
	// 照準の指定がないときは、position の castHeight の高さから forward の向き（水平）へ撃つ。
	void Update(const CharacterInput& input, const Vector3& position, const Vector3& forward, float deltaTime, bool canAct);

	MagicBase* GetCurrentMagic() const;
	void SelectMagic(int index);

private:
	MagicCastContext MakeContext(const CharacterInput& input, const Vector3& position, const Vector3& forward, float deltaTime) const;
	// ボタンの状態に応じて魔法の OnPress / OnHold / OnRelease を呼ぶ。
	static void UpdateButton(MagicBase& magic, const ButtonState& button, const MagicCastContext& context, bool& isHolding);
	void CancelHolding();
	void CancelSubHolding();

	std::vector<std::unique_ptr<MagicBase>> m_magics;
	std::unique_ptr<MagicBase> m_subMagic;
	IGameObject* m_owner = nullptr;
	Settings m_settings;
	Faction m_faction = Faction::Neutral;
	int m_currentIndex = 0;
	bool m_isHolding = false;     // 発射ボタンを押して魔法を使っている最中か。
	bool m_isSubHolding = false;  // サブの発射ボタンを押してサブの魔法を使っている最中か。
};
```

`MagicCaster.cpp` に次を書きます。

<!-- new: ../../../Game/Source/Magic/MagicCaster.cpp -->

```cpp
#include "stdafx.h"
#include "MagicCaster.h"
#include "CharacterInput.h"

void MagicCaster::Init(IGameObject* owner, Faction faction, const Settings& settings)
{
	m_owner = owner;
	m_faction = faction;
	m_settings = settings;
}

void MagicCaster::AddMagic(std::unique_ptr<MagicBase> magic)
{
	if (magic != nullptr) {
		m_magics.push_back(std::move(magic));
	}
}

void MagicCaster::SetSubMagic(std::unique_ptr<MagicBase> magic)
{
	CancelSubHolding();
	m_subMagic = std::move(magic);
}

MagicBase* MagicCaster::GetCurrentMagic() const
{
	if (m_magics.empty()) {
		return nullptr;
	}
	return m_magics[m_currentIndex].get();
}

void MagicCaster::SelectMagic(int index)
{
	if (m_magics.empty()) {
		return;
	}
	const int count = static_cast<int>(m_magics.size());
	// 端まで行ったら反対側に回り込む。
	const int newIndex = ((index % count) + count) % count;
	if (newIndex == m_currentIndex) {
		return;
	}
	CancelHolding();
	m_currentIndex = newIndex;
}

void MagicCaster::Update(const CharacterInput& input, const Vector3& position, const Vector3& forward, float deltaTime, bool canAct)
{
	// 選んでいない魔法もクールダウンは進める。
	for (auto& magic : m_magics) {
		magic->Update(deltaTime);
	}
	if (m_subMagic != nullptr) {
		m_subMagic->Update(deltaTime);
	}

	if (!canAct) {
		CancelHolding();
		CancelSubHolding();
		return;
	}

	if (input.magicSwitch != 0) {
		SelectMagic(m_currentIndex + input.magicSwitch);
	}

	const MagicCastContext context = MakeContext(input, position, forward, deltaTime);
	if (MagicBase* magic = GetCurrentMagic()) {
		UpdateButton(*magic, input.fire, context, m_isHolding);
	}
	if (m_subMagic != nullptr) {
		UpdateButton(*m_subMagic, input.subFire, context, m_isSubHolding);
	}
}

void MagicCaster::UpdateButton(MagicBase& magic, const ButtonState& button, const MagicCastContext& context, bool& isHolding)
{
	if (button.isTrigger) {
		isHolding = true;
		magic.OnPress(context);
	}
	if (isHolding && button.isPress) {
		magic.OnHold(context);
	}
	if (isHolding && !button.isPress) {
		isHolding = false;
		magic.OnRelease(context);
	}
}

MagicCastContext MagicCaster::MakeContext(const CharacterInput& input, const Vector3& position, const Vector3& forward, float deltaTime) const
{
	MagicCastContext context;
	context.caster = m_owner;
	context.faction = m_faction;
	context.deltaTime = deltaTime;

	if (input.hasAim) {
		context.origin = input.aimOrigin;
		context.direction = input.aimDirection;
	}
	else {
		context.origin = position;
		context.origin.y += m_settings.castHeight;
		context.direction = forward;
		context.direction.y = 0.0f;
	}
	if (context.direction.LengthSq() <= FLT_EPSILON) {
		context.direction = Vector3::AxisZ;
	}
	context.direction.Normalize();
	context.origin += context.direction * m_settings.castForward;
	return context;
}

void MagicCaster::CancelHolding()
{
	if (!m_isHolding) {
		return;
	}
	m_isHolding = false;
	if (MagicBase* magic = GetCurrentMagic()) {
		magic->OnCancel();
	}
}

void MagicCaster::CancelSubHolding()
{
	if (!m_isSubHolding) {
		return;
	}
	m_isSubHolding = false;
	if (m_subMagic != nullptr) {
		m_subMagic->OnCancel();
	}
}
```

- **撃つ位置と向きは `MakeContext()` で決める。** 入力に照準があれば（一人称・AI）その位置と向き、なければキャラクターの足元から `castHeight` の高さで正面へ撃ちます。どちらも、自分のカプセルに当たらないよう `castForward` だけ前に出します。
- **行動できない（凍結中など）ときは撃たない。** 押している途中なら `OnCancel()` を呼んで、溜めなどを捨てさせます。
- **選んでいない魔法もクールダウンは進める。**

**確かめ方:** ソリューションのビルドが通る。
