作成日: 2026-11-08
設計: [陣営と当たり判定の設計](陣営と当たり判定の設計.md)

キャラクターの陣営（Player・味方・中立・敵）と、弾などに当ててもらうための当たり判定（`HitTarget`）を作り、キャラクターに持たせるまでの手順です。

**出来上がるもの:** Player は陣営 Player、Enemy は陣営 Enemy で作られ、それぞれの当たり判定が一覧（`HitTargetRegistry`）に登録されます。弾や AI は、この一覧から相手を探せるようになります（[弾の手順](../魔法/弾の手順.md)・[NPCのAIの手順](NPCのAIの手順.md)）。

**前提:**

- Visual Studio 2022 で `Action/Game/Game.sln` を開き、構成は **Debug / x64** でビルドする。
- 先に [キャラクターの構成の手順](キャラクターの構成の手順.md) を済ませておく（順番は [README の「ゲームを作る順番」](../../README.md#ゲームを作る順番)）。
- 作業を始める前に、main からブランチを切る。

| 手順 | 足すもの | 確かめ方 |
|---|---|---|
| 1 | 陣営と、敵対しているかの判定 | ビルドが通る |
| 2 | 当たり判定 `HitTarget` と、その一覧 `HitTargetRegistry` | ビルドが通る |

```cpp
#pragma once

// キャラクターの陣営。弾の当たり判定やAIの狙う相手の判断に使う。
enum class Faction
{
	Player,   // プレイヤー本人。
	Ally,     // プレイヤーの味方NPC。
	Neutral,  // 中立NPC。誰とも敵対しない（弾も当たらない）。
	Enemy,    // 敵NPC。
};

// a と b が敵対しているか。弾は敵対している相手にだけ当たり、AIは敵対している相手だけを狙う。
inline bool IsHostile(Faction a, Faction b)
{
	if (a == Faction::Neutral || b == Faction::Neutral) {
		return false;
	}
	const bool isAFriendly = a == Faction::Player || a == Faction::Ally;
	const bool isBFriendly = b == Faction::Player || b == Faction::Ally;
	return isAFriendly != isBFriendly;
}
```

中立はだれとも敵対しません（弾も当たらない）。Player と味方は同じ側、敵は反対の側です。

**確かめ方:** ソリューションのビルドが通る。

```cpp
#pragma once

#include "Faction.h"

// 何かに当たったときに受け手へ渡す情報。
struct HitInfo
{
	const IGameObject* attacker = nullptr;  // 撃ったキャラクター。削除済みの可能性があるので比較にだけ使う。
	Faction attackerFaction = Faction::Neutral;
	Vector3 position = Vector3::Zero;       // 当たった位置。
	Vector3 direction = Vector3::AxisZ;     // 当たったものの進行方向。
};

// 弾などに当ててもらうための当たり判定。キャラクターがメンバーとして持つ。
// Init() で HitTargetRegistry に登録され、デストラクタで登録が解除される。
// 弾やAIは HitTargetRegistry を通して相手を探すので、相手のクラスを知らなくてよい。
class HitTarget : public Noncopyable
{
public:
	using OnHitFunc = std::function<void(const HitInfo&)>;

	HitTarget() = default;
	~HitTarget();

	// centerHeight: 足元から当たり判定の中心までの高さ。radius: 当たり判定の球の半径。
	void Init(IGameObject* owner, Faction faction, float radius, float centerHeight);
	// 持ち主が毎フレーム足元の位置を渡す。
	void SetPosition(const Vector3& footPosition)
	{
		m_footPosition = footPosition;
	}
	void SetFaction(Faction faction)
	{
		m_faction = faction;
	}
	// 当たったときに持ち主へ知らせる処理（ひるみ、反撃など）。
	void SetOnHit(OnHitFunc onHit)
	{
		m_onHit = std::move(onHit);
	}

	// 弾などが当たったときに呼ぶ。
	void NotifyHit(const HitInfo& hitInfo) const;

	Vector3 GetCenter() const;
	const Vector3& GetFootPosition() const
	{
		return m_footPosition;
	}
	float GetRadius() const
	{
		return m_radius;
	}
	Faction GetFaction() const
	{
		return m_faction;
	}
	const IGameObject* GetOwner() const
	{
		return m_owner;
	}
	// 持ち主が削除待ちになっていたら当たらないようにする。
	bool IsActive() const;

private:
	IGameObject* m_owner = nullptr;
	OnHitFunc m_onHit;
	Vector3 m_footPosition = Vector3::Zero;
	Faction m_faction = Faction::Neutral;
	float m_radius = 0.0f;
	float m_centerHeight = 0.0f;
	bool m_isRegistered = false;
};

// 全キャラクターの当たり判定の一覧。弾の命中判定とAIの索敵に使う。
class HitTargetRegistry
{
public:
	static void Register(HitTarget* target);
	static void Unregister(HitTarget* target);

	// from→to を半径 radius の球が通るとき、attackerFaction と敵対する相手のうち最初に当たるものを返す。
	// ignoreOwner の当たり判定は無視する（撃った本人に当たらないように）。見つからなければ nullptr。
	static HitTarget* FindHitOnSegment(
		const Vector3& from, const Vector3& to, float radius,
		Faction attackerFaction, const IGameObject* ignoreOwner,
		Vector3& hitPosition);

	// position から maxDistance 以内で、faction と敵対する相手のうち一番近いものを返す。
	static HitTarget* FindNearestHostile(
		const Vector3& position, float maxDistance,
		Faction faction, const IGameObject* ignoreOwner);

	// 登録されている当たり判定をデバッグ表示用に返す。
	static const std::vector<HitTarget*>& GetTargets();

private:
	static std::vector<HitTarget*>& Targets();
};
```

```cpp
#include "stdafx.h"
#include "HitTarget.h"

namespace
{
	// 線分 from→to 上で point に最も近い点を返す。
	Vector3 ClosestPointOnSegment(const Vector3& from, const Vector3& to, const Vector3& point)
	{
		Vector3 segment = to - from;
		const float lengthSq = segment.LengthSq();
		if (lengthSq <= FLT_EPSILON) {
			return from;
		}
		float t = (point - from).Dot(segment) / lengthSq;
		t = t < 0.0f ? 0.0f : (t > 1.0f ? 1.0f : t);
		return from + segment * t;
	}
}

HitTarget::~HitTarget()
{
	if (m_isRegistered) {
		HitTargetRegistry::Unregister(this);
	}
}

void HitTarget::Init(IGameObject* owner, Faction faction, float radius, float centerHeight)
{
	m_owner = owner;
	m_faction = faction;
	m_radius = radius;
	m_centerHeight = centerHeight;
	if (!m_isRegistered) {
		HitTargetRegistry::Register(this);
		m_isRegistered = true;
	}
}

void HitTarget::NotifyHit(const HitInfo& hitInfo) const
{
	if (m_onHit) {
		m_onHit(hitInfo);
	}
}

Vector3 HitTarget::GetCenter() const
{
	Vector3 center = m_footPosition;
	center.y += m_centerHeight;
	return center;
}

bool HitTarget::IsActive() const
{
	return m_owner != nullptr && !m_owner->IsDead();
}

void HitTargetRegistry::Register(HitTarget* target)
{
	Targets().push_back(target);
}

void HitTargetRegistry::Unregister(HitTarget* target)
{
	std::vector<HitTarget*>& targets = Targets();
	targets.erase(std::remove(targets.begin(), targets.end(), target), targets.end());
}

HitTarget* HitTargetRegistry::FindHitOnSegment(
	const Vector3& from, const Vector3& to, float radius,
	Faction attackerFaction, const IGameObject* ignoreOwner,
	Vector3& hitPosition)
{
	HitTarget* hitTarget = nullptr;
	// 複数に当たる場合は、線分の始点に一番近い相手を選ぶ。
	float nearestFromDistanceSq = FLT_MAX;
	for (HitTarget* target : Targets()) {
		if (!target->IsActive() || target->GetOwner() == ignoreOwner) {
			continue;
		}
		if (!IsHostile(attackerFaction, target->GetFaction())) {
			continue;
		}
		const Vector3 center = target->GetCenter();
		const Vector3 closest = ClosestPointOnSegment(from, to, center);
		const float hitRadius = radius + target->GetRadius();
		if ((center - closest).LengthSq() > hitRadius * hitRadius) {
			continue;
		}
		const float fromDistanceSq = (closest - from).LengthSq();
		if (fromDistanceSq < nearestFromDistanceSq) {
			nearestFromDistanceSq = fromDistanceSq;
			hitTarget = target;
			hitPosition = closest;
		}
	}
	return hitTarget;
}

HitTarget* HitTargetRegistry::FindNearestHostile(
	const Vector3& position, float maxDistance,
	Faction faction, const IGameObject* ignoreOwner)
{
	HitTarget* nearest = nullptr;
	float nearestDistanceSq = maxDistance * maxDistance;
	for (HitTarget* target : Targets()) {
		if (!target->IsActive() || target->GetOwner() == ignoreOwner) {
			continue;
		}
		if (!IsHostile(faction, target->GetFaction())) {
			continue;
		}
		const float distanceSq = (target->GetCenter() - position).LengthSq();
		if (distanceSq <= nearestDistanceSq) {
			nearestDistanceSq = distanceSq;
			nearest = target;
		}
	}
	return nearest;
}

const std::vector<HitTarget*>& HitTargetRegistry::GetTargets()
{
	return Targets();
}

std::vector<HitTarget*>& HitTargetRegistry::Targets()
{
	// 関数内の static にして、他の static 変数との初期化順の問題を避ける。
	static std::vector<HitTarget*> s_targets;
	return s_targets;
}
```

- **当たり判定は球。** 足元の位置に `centerHeight` を足したところが中心です。
- **線分で当たる相手を探す（`FindHitOnSegment()`）。** 弾が1フレームに進む線分と、相手の球が重なるかを見ます。いくつも当たるときは、線分の始点にいちばん近い相手にします。
- **一覧は関数の中の `static` にする（`Targets()`）。** ほかの `static` 変数との初期化の順番の問題を避けるためです。
- **持ち主が削除待ち（`IsDead()`）なら当たらない。**

**確かめ方:** ソリューションのビルドが通る。
